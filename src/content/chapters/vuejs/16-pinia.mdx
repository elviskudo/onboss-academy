Pinia menyimpan state global dengan API yang sederhana dan bertipe baik.

## Membuat Store

```js
// stores/keranjang.js
import { defineStore } from 'pinia'

export const useKeranjang = defineStore('keranjang', {
  state: () => ({
    items: [],
  }),
  getters: {
    jumlah: (state) => state.items.length,
    total: (state) => state.items.reduce((sum, i) => sum + i.harga, 0),
  },
  actions: {
    tambah(produk) {
      this.items.push(produk)
    },
    hapus(id) {
      this.items = this.items.filter((i) => i.id !== id)
    },
  },
})
```

## Setup-style

```js
export const useKeranjang = defineStore('keranjang', () => {
  const items = ref([])
  const total = computed(() => items.value.reduce((s, i) => s + i.harga, 0))

  function tambah(p) { items.value.push(p) }

  return { items, total, tambah }
})
```

## Memakai di Komponen

```vue
<script setup>
import { useKeranjang } from '@/stores/keranjang'
import { storeToRefs } from 'pinia'

const keranjang = useKeranjang()
const { items, total } = storeToRefs(keranjang)
</script>

<template>
  <p>{{ items.length }} item — Rp{{ total }}</p>
  <button @click="keranjang.tambah({ id: 1, harga: 25000 })">Tambah</button>
</template>
```

`storeToRefs` menjaga reaktivitas saat mendestrukturi state/getters.

## Action Async

```js
actions: {
  async muat() {
    this.loading = true
    try {
      this.items = await api.daftar()
    } finally {
      this.loading = false
    }
  },
}
```

<div class="note">💡 Pakai Pinia untuk state yang benar-benar lintas halaman (keranjang, sesi user, preferensi). State lokal satu komponen cukup dengan `ref`.</div>
