Composable adalah cara Vue memakai ulang logika berstate. Mirip custom hook di React.

## Contoh: useLocalStorage

```js
// composables/useLocalStorage.js
import { ref, watch } from 'vue'

export function useLocalStorage(kunci, awal) {
  const data = ref(
    JSON.parse(localStorage.getItem(kunci) ?? 'null') ?? awal
  )

  watch(data, (nilai) => {
    localStorage.setItem(kunci, JSON.stringify(nilai))
  }, { deep: true })

  return { data }
}
```

Pemakaian:

```vue
<script setup>
import { useLocalStorage } from '@/composables/useLocalStorage'

const { data: tema } = useLocalStorage('tema', 'terang')
</script>

<template>
  <button @click="tema = tema === 'terang' ? 'gelap' : 'terang'">
    Tema: {{ tema }}
  </button>
</template>
```

## Contoh: useFetch

```js
export function useFetch(url) {
  const data = ref(null)
  const error = ref('')
  const loading = ref(true)

  watchEffect(async (onCleanup) => {
    const controller = new AbortController()
    onCleanup(() => controller.abort())
    loading.value = true

    try {
      const res = await fetch(url.value, { signal: controller.signal })
      data.value = await res.json()
    } catch (e) {
      if (e.name !== 'AbortError') error.value = e.message
    } finally {
      loading.value = false
    }
  })

  return { data, error, loading }
}
```

## Aturan

- Panggil di dalam `setup`.
- Gunakan awalan `use`.
- Kembalikan nilai reaktif, bukan hasil hitung sekali jalan.

<div class="note">💡 Composable membuat komponen tetap fokus pada tampilan. Kalau ada blok `ref` + `watch` yang sama di tiga komponen, itu calon composable.</div>
