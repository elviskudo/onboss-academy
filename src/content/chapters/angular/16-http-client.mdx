HttpClient adalah cara standar Angular berkomunikasi dengan server. Responsnya berupa Observable yang bertipe.

## Menyiapkan

```typescript
// app.config.ts
import { ApplicationConfig } from "@angular/core";
import { provideHttpClient, withInterceptors } from "@angular/common/http";

export const appConfig: ApplicationConfig = {
  providers: [provideHttpClient(withInterceptors([authInterceptor]))],
};
```

## Request Dasar

```typescript
import { HttpClient } from "@angular/common/http";
import { inject } from "@angular/core";

export class ProdukService {
  private http = inject(HttpClient);

  daftar() {
    return this.http.get<Produk[]>("/api/produk");
  }

  detail(id: number) {
    return this.http.get<Produk>(`/api/produk/${id}`);
  }

  simpan(data: Produk) {
    return this.http.post<Produk>("/api/produk", data);
  }

  hapus(id: number) {
    return this.http.delete(`/api/produk/${id}`);
  }
}
```

Generic `<Produk[]>` memberi tipe pada hasil — TypeScript membantu mencegah salah pakai field.

## Memakai di Komponen

```typescript
export class DaftarComponent implements OnInit {
  private api = inject(ProdukService);
  produk: Produk[] = [];
  error = "";

  ngOnInit() {
    this.api.daftar().subscribe({
      next: (data) => this.produk = data,
      error: (e) => this.error = "Gagal memuat: " + e.message,
    });
  }
}
```

Atau pakai async pipe di template agar Angular yang mengelola subscription.

## Interceptor

```typescript
import { HttpInterceptorFn } from "@angular/common/http";

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = localStorage.getItem("token");
  const request = token
    ? req.clone({ setHeaders: { Authorization: `Bearer ${token}` } })
    : req;
  return next(request);
};
```

## Error Global

```typescript
return next(req).pipe(
  catchError((err) => {
    console.error(err.status, err.message);
    return throwError(() => err);
  })
);
```

<div class="note">💡 Simpan request di service, bukan langsung di komponen. Saat endpoint berubah, kamu cukup mengubah satu tempat.</div>
