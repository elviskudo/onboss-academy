Proyek penutup ini merangkum materia Go yang sudah kamu pelajari: struct, JSON, HTTP, context, goroutine, error, dan testing.

## Rancangan

```
main.go        -> entry point
model.go       -> struct + tag JSON
client.go      -> HTTP + context
processor.go   -> logika murni
client_test.go -> unit test
```

## Model

```go
type Artikel struct {
    ID     int    `json:"id"`
    Judul  string `json:"judul"`
    Penulis string `json:"penulis,omitempty"`
}
```

## Klien dengan Context

```go
func Ambil(ctx context.Context, url string) ([]Artikel, error) {
    req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
    if err != nil { return nil, err }

    resp, err := http.DefaultClient.Do(req)
    if err != nil { return nil, err }
    defer resp.Body.Close()

    var data []Artikel
    if err := json.NewDecoder(resp.Body).Decode(&data); err != nil {
        return nil, fmt.Errorf("decode %s: %w", url, err)
    }
    return data, nil
}
```

## Logika Murni

```go
func JudulUnik(a []Artikel) []string {
    terlihat := map[string]bool{}
    var hasil []string
    for _, art := range a {
        if !terlihat[art.Judul] {
            terlihat[art.Judul] = true
            hasil = append(hasil, art.Judul)
        }
    }
    return hasil
}
```

Fungsi ini tidak tahu soal HTTP sama sekali — mudah diuji.

## Konkurensi Terbatas

```go
var wg sync.WaitGroup
for _, url := range urls {
    wg.Add(1)
    go func(u string) {
        defer wg.Done()
        data, err := Ambil(ctx, u)
        // kumpulkan hasil dengan channel / mutex
    }(url)
}
wg.Wait()
```

## Test

```go
func TestJudulUnik(t *testing.T) {
    input := []Artikel{{Judul: "Go"}, {Judul: "Go"}, {Judul: "Clean Code"}}
    got := JudulUnik(input)
    if len(got) != 2 { t.Fatalf("dapat %d, ingin 2", len(got)) }
}
```

<div class="note">💡 Kode yang bisa diuji tanpa jaringan adalah kode yang rancangannya sudah benar. Pisahkan "mengambil data" dari "mengolah data" sejak awal.</div>

import (
    "context"
    "fmt"
    "time"
)

type Artikel struct {
    ID     int
    Judul  string
    Penulis string
}

func judulUnik(data []Artikel) []string {
    terlihat := map[string]bool{}
    var hasil []string
    for _, a := range data {
        if !terlihat[a.Judul] {
            terlihat[a.Judul] = true
            hasil = append(hasil, a.Judul)
        }
    }

func ambil(ctx context.Context, id int) (Artikel, error) {
    select {
    case <-time.After(10 * time.Millisecond):
        return Artikel{ID: id, Judul: fmt.Sprintf("Artikel %d", id)}, nil
    case <-ctx.Done():
        return Artikel{}, ctx.Err()
    }
}

func main() {
    ctx, cancel := context.WithTimeout(context.Background(), time.Second)
    defer cancel()

    a, err := ambil(ctx, 1)
    if err != nil { fmt.Println(err); return }
    fmt.Println("Ambil:", a)

    data := []Artikel{
        {ID: 1, Judul: "Go", Penulis: "Rani"},
        {ID: 2, Judul: "Go"},
        {ID: 3, Judul: "Clean Code"},
    }
    fmt.Println("Judul unik:", judulUnik(data))
}`} />
