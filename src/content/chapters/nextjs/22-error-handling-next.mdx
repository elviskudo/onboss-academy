Error yang tidak tertangani membuat halaman kosong. Next.js menyediakan boundary di level segmen.

## error.tsx

```tsx
"use client";

import { useEffect } from "react";

export default function Error({
  error,
  reset,
}: {
  error: Error;
  reset: () => void;
}) {
  useEffect(() => console.error(error), [error]);

  return (
    <div className="error">
      <h2>Ada yang salah.</h2>
      <p>{error.message}</p>
      <button onClick={reset}>Coba lagi</button>
    </div>
  );
}
```

Taruh di `app/produk/error.tsx` untuk melindungi segmen tersebut saja.

## not-found.tsx

```tsx
// app/not-found.tsx
export default function NotFound() {
  return (
    <div>
      <h1>404 — Halaman tidak ditemukan</h1>
      <a href="/">Kembali ke beranda</a>
    </div>
  );
}
```

Panggil `notFound()` bila data tidak ada:

```ts
const produk = await ambilProduk(id);
if (!produk) notFound();
```

## Error di Server Action

```ts
"use server";

export async function simpan(prevState, formData) {
  try {
    await db.simpan(formData);
    return { pesan: "Berhasil" };
  } catch (e) {
    return { pesan: "Gagal menyimpan, coba lagi." };
  }
}
```
