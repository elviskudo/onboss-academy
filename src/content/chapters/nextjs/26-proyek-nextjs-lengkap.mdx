Penutup seri ini: merangkai semua konsep menjadi aplikasi katalog produk siap pakai.

## Struktur

```
app/
  layout.tsx
  page.tsx
  produk/
    page.tsx
    [id]/page.tsx
    action.ts
    loading.tsx
    error.tsx
  api/produk/route.ts
components/
  KartuProduk.tsx
lib/
  db.ts
```

## Halaman Daftar

```tsx
export const revalidate = 60;

export default async function HalamanProduk() {
  const produk = await ambilProduk();

  return (
    <main>
      <h1>Katalog Produk</h1>
      <div className="grid">
        {produk.map((p) => (
          <a key={p.id} href={`/produk/${p.id}`}>
            <KartuProduk nama={p.nama} harga={p.harga} />
          </a>
        ))}
      </div>
    </main>
  );
}
```

## Detail

```tsx
export async function generateMetadata({ params }) {
  const p = await ambilProduk(params.id);
  return { title: p?.nama ?? "Produk" };
}

export default async function Detail({ params }) {
  const p = await ambilProduk(params.id);
  if (!p) notFound();

  return (
    <article>
      <h1>{p.nama}</h1>
      <p>Rp{p.harga.toLocaleString("id-ID")}</p>
      <form action={tambahKeKeranjang}>
        <input type="hidden" name="id" value={p.id} />
        <button type="submit">Tambah ke keranjang</button>
      </form>
    </article>
  );
}
```
