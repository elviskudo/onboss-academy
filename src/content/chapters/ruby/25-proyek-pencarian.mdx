## Modul Analisis

```ruby
# lib/analyzer.rb
module Analyzer
  def self.muat(path)
    CSV.read(path, headers: true)
  rescue Errno::ENOENT
    []
  end

  def self.ringkas(baris)
    baris.group_by { |r| r["kota"] }
         .transform_values(&:size)
  end

  def self.cari(baris, kata)
    pola = Regexp.new(Regexp.escape(kata), Regexp::IGNORECASE)
    baris.select { |r| r["nama"]&.match?(pola) }
  end
end
```

## Memakai Modul

```ruby
require_relative "lib/analyzer"

data = [
  { "nama" => "Rani", "kota" => "Bandung" },
  { "nama" => "Dimas", "kota" => "Jakarta" }
]

p Analyzer.ringkas(data)
p Analyzer.cari(data, "ran")
```

## Test

```ruby
require "minitest/autorun"
require_relative "lib/analyzer"

class TesAnalyzer < Minitest::Test
  def test_ringkas
    data = [{ "kota" => "Bandung" }, { "kota" => "Bandung" }]
    assert_equal({ "Bandung" => 2 }, Analyzer.ringkas(data))
  end
end
```
